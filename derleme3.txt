          package com.example.chesshelper

          import android.graphics.Bitmap
          import kotlin.math.abs

          // Her kare 24x24'e küçültülür, kenardan 3 piksel içerideki 18x18 alan analiz edilir
          private const val CELL = 24
          private const val MARGIN = 3
          private const val INNER = CELL - 2 * MARGIN
          private const val OCCUPIED_RATIO = 0.10f   // karenin en az %10'u doluysa taş var say
          private const val MIN_MATCH = 0.35f        // okurken şablon eşleşme alt sınırı (IoU)
          private const val SAMPLE_MATCH = 0.60f     // kalibrasyonda aynı tür taşlar birbirine bu kadar benzemeli
          private const val TINT_DIFF = 35           // vurgulu (son hamle) kare: köşe rengine yakın pikseller arka plan
          private const val PIECE_LETTERS = "PRNBQK" // şablon indeksleri

          /** Okunan pozisyon. grid[rank][file]: rank 0 = 1. sıra, file 0 = a. Büyük harf = beyaz. */
          class ChessPosition(private val grid: Array<Array<Char?>>) {

              fun placement(): String {
                  val sb = StringBuilder()
                  for (rank in 7 downTo 0) {
                      var empty = 0
                      for (file in 0 until 8) {
                          val c = grid[rank][file]
                          if (c == null) {
                              empty++
                          } else {
                              if (empty > 0) { sb.append(empty); empty = 0 }
                              sb.append(c)
                          }
                      }
                      if (empty > 0) sb.append(empty)
                      if (rank > 0) sb.append('/')
                  }
                  return sb.toString()
              }

              /** Rok hakları, şah ve kale başlangıç karelerinde mi diye tahmin edilir. */
              fun fen(sideToMove: Char): String {
                  val castling = buildString {
                      if (grid[0][4] == 'K' && grid[0][7] == 'R') append('K')
                      if (grid[0][4] == 'K' && grid[0][0] == 'R') append('Q')
                      if (grid[7][4] == 'k' && grid[7][7] == 'r') append('k')
                      if (grid[7][4] == 'k' && grid[7][0] == 'r') append('q')
                  }.ifEmpty { "-" }
                  return "${placement()} $sideToMove $castling - 0 1"
              }

              /** Verilen kare hariç iki pozisyon aynı mı? (öneri kutusunun altındaki kareyi saymamak için) */
              fun sameExcept(other: ChessPosition, skipFile: Int, skipRank: Int): Boolean {
                  for (r in 0 until 8) for (f in 0 until 8) {
                      if (r == skipRank && f == skipFile) continue
                      if (grid[r][f] != other.grid[r][f]) return false
                  }
                  return true
              }

              companion object {
                  /** Son hamleyi kimin yaptığı: karesi boşalan taşın rengi. Belirsizse null. */
                  fun moverColor(prev: ChessPosition, cur: ChessPosition): Char? {
                      var white = false
                      var black = false
                      for (r in 0 until 8) for (f in 0 until 8) {
                          val p = prev.grid[r][f]
                          if (p != null && cur.grid[r][f] == null) {
                              if (p.isUpperCase()) white = true else black = true
                          }
                      }
                      return when {
                          white && !black -> 'w'
                          black && !white -> 'b'
                          else -> null
                      }
                  }
              }
          }

          /**
           * Ekran görüntüsünden tahtayı okur.
           *
           * Kalibrasyon (oyun BAŞLANGIÇ konumundayken): 3-6. sıralardaki boş karelerin renklerinden
           * "arka plan paleti" öğrenilir (dokulu çimen tahtalar dahil), 1-2 ve 7-8. sıralardaki taşlardan
           * da şekil şablonları çıkarılır. Sonra her kare paletin dışında kalan piksellerden oluşan
           * siluetle şablonlara karşılaştırılır. Taşın rengi (beyaz/siyah) siluetin parlaklığından anlaşılır.
           */
          class BoardRecognizer {

              private class Cell(val mask: BooleanArray, val bright: Float, val fill: Float) {
                  val occupied: Boolean get() = fill >= OCCUPIED_RATIO
              }

              private var palette: BooleanArray? = null                    // 32*32*32 renk tablosu
              private var templates: Array<Array<BooleanArray>>? = null    // [0 beyaz / 1 siyah][parça]
              private var brightThreshold = 0.5f

              /** true: siyah altta oynuyor */
              var flipped = false
                  private set

              val isCalibrated: Boolean get() = templates != null && palette != null

              private fun imageRow(rank: Int, fl: Boolean) = if (fl) rank else 7 - rank
              private fun imageCol(file: Int, fl: Boolean) = if (fl) 7 - file else file

              /** Tahta başlangıç konumundaysa paleti ve şablonları öğrenir, true döner. */
              fun calibrate(board: Bitmap): Boolean {
                  val px = downscale(board)
                  val pal = buildPalette(px)
                  val cells = Array(8) { row -> Array(8) { col -> analyzeCell(px, pal, row, col) } }

                  // 1) 1,2,7,8. sıralar dolu, 3-6. sıralar boş olmalı
                  for (row in 0 until 8) for (col in 0 until 8) {
                      val shouldBeOccupied = row <= 1 || row >= 6
                      if (cells[row][col].occupied != shouldBeOccupied) return false
                  }

                  // 2) Üst ve alt taraf renkleri belirgin şekilde farklı olmalı
                  var top = 0f
                  var bottom = 0f
                  for (col in 0 until 8) {
                      top += cells[0][col].bright + cells[1][col].bright
                      bottom += cells[6][col].bright + cells[7][col].bright
                  }
                  top /= 16f
                  bottom /= 16f
                  if (abs(top - bottom) < 0.15f) return false

                  val fl = bottom < top // üst taraf daha açıksa beyaz üsttedir, yani siyah altta

                  // 3) Şablonlar
                  val samples = Array(2) { Array(6) { ArrayList<BooleanArray>() } }
                  val backRank = intArrayOf(1, 2, 3, 4, 5, 3, 2, 1) // R N B Q K B N R (a..h)
                  for (color in 0..1) {
                      val backR = if (color == 0) 0 else 7
                      val pawnR = if (color == 0) 1 else 6
                      for (file in 0 until 8) {
                          samples[color][backRank[file]].add(cells[imageRow(backR, fl)][imageCol(file, fl)].mask)
                          samples[color][0].add(cells[imageRow(pawnR, fl)][imageCol(file, fl)].mask)
                      }
                  }
                  val tpl = Array(2) { color ->
                      Array(6) { type ->
                          val list = samples[color][type]
                          BooleanArray(INNER * INNER) { i ->
                              var c = 0
                              for (m in list) if (m[i]) c++
                              c * 2 > list.size
                          }
                      }
                  }

                  // 4) Aynı türden taşlar (8 piyon, 2 kale...) birbirine benzemeli, yoksa bu gerçek bir başlangıç konumu değil
                  for (color in 0..1) for (type in 0 until 6) {
                      for (m in samples[color][type]) {
                          if (iou(m, tpl[color][type]) < SAMPLE_MATCH) return false
                      }
                  }

                  flipped = fl
                  brightThreshold = (top + bottom) / 2f
                  palette = pal
                  templates = tpl
                  return true
              }

              /**
               * Pozisyonu okur. Okunamıyorsa (animasyon, tanınmayan taş vb.) null döner.
               * skipFile/skipRank verilirse o kare okunmaz (üstünde öneri kutusu var).
               */
              fun read(board: Bitmap, skipFile: Int = -1, skipRank: Int = -1): ChessPosition? {
                  val tpl = templates ?: return null
                  val pal = palette ?: return null
                  val px = downscale(board)
                  val fl = flipped
                  val grid = Array(8) { arrayOfNulls<Char>(8) }
                  var whiteKings = 0
                  var blackKings = 0

                  for (rank in 0 until 8) for (file in 0 until 8) {
                      if (file == skipFile && rank == skipRank) continue
                      val cell = analyzeCell(px, pal, imageRow(rank, fl), imageCol(file, fl))
                      if (!cell.occupied) continue

                      val color = if (cell.bright > brightThreshold) 0 else 1
                      var best = -1
                      var bestScore = 0f
                      for (t in 0 until 6) {
                          val s = iou(cell.mask, tpl[color][t])
                          if (s > bestScore) { bestScore = s; best = t }
                      }
                      if (best < 0 || bestScore < MIN_MATCH) return null
                      if (best == 0 && (rank == 0 || rank == 7)) return null // piyon 1. veya 8. sırada olamaz

                      val ch = PIECE_LETTERS[best]
                      grid[rank][file] = if (color == 0) ch else ch.lowercaseChar()
                      if (best == 5) { if (color == 0) whiteKings++ else blackKings++ }
                  }

                  if (skipFile >= 0) {
                      if (whiteKings > 1 || blackKings > 1) return null
                  } else {
                      if (whiteKings != 1 || blackKings != 1) return null
                  }
                  return ChessPosition(grid)
              }

              private fun iou(a: BooleanArray, b: BooleanArray): Float {
                  var inter = 0
                  var union = 0
                  for (i in a.indices) {
                      if (a[i] && b[i]) inter++
                      if (a[i] || b[i]) union++
                  }
                  return if (union == 0) 0f else inter.toFloat() / union
              }

              private fun binIndex(p: Int): Int =
                  (((p shr 19) and 31) shl 10) or (((p shr 11) and 31) shl 5) or ((p shr 3) and 31)

              /** Tahtayı 192x192'ye küçültür (kutu ortalaması). Her piksel 0xRRGGBB. */
              private fun downscale(board: Bitmap): IntArray {
                  val w = board.width
                  val h = board.height
                  val src = IntArray(w * h)
                  board.getPixels(src, 0, w, 0, 0, w, h)
                  val side = 8 * CELL
                  val out = IntArray(side * side)
                  for (oy in 0 until side) {
                      val y0 = oy * h / side
                      val y1 = maxOf(y0 + 1, (oy + 1) * h / side)
                      for (ox in 0 until side) {
                          val x0 = ox * w / side
                          val x1 = maxOf(x0 + 1, (ox + 1) * w / side)
                          var r = 0
                          var g = 0
                          var b = 0
                          for (y in y0 until y1) {
                              val row = y * w
                              for (x in x0 until x1) {
                                  val p = src[row + x]
                                  r += (p shr 16) and 0xFF
                                  g += (p shr 8) and 0xFF
                                  b += p and 0xFF
                              }
                          }
                          val n = (y1 - y0) * (x1 - x0)
                          out[oy * side + ox] = ((r / n) shl 16) or ((g / n) shl 8) or (b / n)
                      }
                  }
                  return out
              }

              /** 3-6. sıralar (görüntüde 2..5. satırlar) başlangıçta boştur: renklerini "arka plan" olarak öğren. */
              private fun buildPalette(px: IntArray): BooleanArray {
                  val side = 8 * CELL
                  val base = BooleanArray(32 * 32 * 32)
                  for (row in 2..5) for (y in 0 until CELL) for (x in 0 until side) {
                      base[binIndex(px[(row * CELL + y) * side + x])] = true
                  }
                  // Komşu renk kutularını da kabul et (küçük renk sapmaları için)
                  val out = BooleanArray(32 * 32 * 32)
                  for (r in 0 until 32) for (g in 0 until 32) for (b in 0 until 32) {
                      if (!base[(r shl 10) or (g shl 5) or b]) continue
                      for (dr in -1..1) for (dg in -1..1) for (db in -1..1) {
                          val rr = r + dr
                          val gg = g + dg
                          val bb = b + db
                          if (rr in 0..31 && gg in 0..31 && bb in 0..31) {
                              out[(rr shl 10) or (gg shl 5) or bb] = true
                          }
                      }
                  }
                  return out
              }

              private fun analyzeCell(px: IntArray, pal: BooleanArray, row: Int, col: Int): Cell {
                  val side = 8 * CELL
                  val y0 = row * CELL
                  val x0 = col * CELL

                  // Köşelerdeki dört 2x2 blok (16 piksel): kare vurgulu mu diye bakılır
                  val cr = IntArray(16)
                  val cg = IntArray(16)
                  val cb = IntArray(16)
                  var n = 0
                  var inPal = 0
                  for (cy in intArrayOf(0, CELL - 2)) for (cx in intArrayOf(0, CELL - 2)) {
                      for (dy in 0..1) for (dx in 0..1) {
                          val p = px[(y0 + cy + dy) * side + x0 + cx + dx]
                          cr[n] = (p shr 16) and 0xFF
                          cg[n] = (p shr 8) and 0xFF
                          cb[n] = p and 0xFF
                          if (pal[binIndex(p)]) inPal++
                          n++
                      }
                  }
                  val tinted = inPal < 8 // köşelerin yarısından fazlası bilinen renklerde değil
                  var mr = 0
                  var mg = 0
                  var mb = 0
                  if (tinted) {
                      cr.sort(); cg.sort(); cb.sort()
                      mr = (cr[7] + cr[8]) / 2
                      mg = (cg[7] + cg[8]) / 2
                      mb = (cb[7] + cb[8]) / 2
                  }

                  val total = INNER * INNER
                  val raw = BooleanArray(total)
                  val lum = IntArray(total)
                  for (y in 0 until INNER) for (x in 0 until INNER) {
                      val p = px[(y0 + MARGIN + y) * side + x0 + MARGIN + x]
                      val r = (p shr 16) and 0xFF
                      val g = (p shr 8) and 0xFF
                      val b = p and 0xFF
                      var isBg = pal[binIndex(p)]
                      if (!isBg && tinted) {
                          isBg = abs(r - mr) + abs(g - mg) + abs(b - mb) <= TINT_DIFF
                      }
                      val i = y * INNER + x
                      raw[i] = !isBg
                      lum[i] = (299 * r + 587 * g + 114 * b) / 1000
                  }

                  // Siluet: kenardan ulaşılabilen arka planı işaretle, geri kalan her şey taştır.
                  val outside = BooleanArray(total)
                  val queue = IntArray(total)
                  var head = 0
                  var tail = 0
                  fun seed(i: Int) {
                      if (!raw[i] && !outside[i]) { outside[i] = true; queue[tail++] = i }
                  }
                  for (k in 0 until INNER) {
                      seed(k)
                      seed((INNER - 1) * INNER + k)
                      seed(k * INNER)
                      seed(k * INNER + INNER - 1)
                  }
                  while (head < tail) {
                      val i = queue[head++]
                      val y = i / INNER
                      val x = i % INNER
                      if (y > 0) seed(i - INNER)
                      if (y < INNER - 1) seed(i + INNER)
                      if (x > 0) seed(i - 1)
                      if (x < INNER - 1) seed(i + 1)
                  }

                  val mask = BooleanArray(total) { !outside[it] }
                  var count = 0
                  var brightCount = 0
                  for (i in mask.indices) {
                      if (mask[i]) {
                          count++
                          if (lum[i] > 128) brightCount++
                      }
                  }
                  val bright = if (count == 0) 0f else brightCount.toFloat() / count
                  return Cell(mask, bright, count.toFloat() / total)
              }
          }

              private var notificationBuilder: NotificationCompat.Builder? = null
              private var lastStatus = ""

              override fun onBind(intent: Intent?): IBinder? = null

              override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
                  createNotificationChannel()

                  val builder = NotificationCompat.Builder(this, "CHESS_CHANNEL")
                      .setContentTitle("Satranç Yardımcısı")
                      .setContentText("Başlatılıyor...")
                      .setSmallIcon(android.R.drawable.ic_dialog_info)
                      .setOnlyAlertOnce(true)
                      .setOngoing(true)
                  notificationBuilder = builder
                  val notification = builder.build()

                  if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                      startForeground(1, notification, ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PROJECTION)
                  } else {
                      startForeground(1, notification)
                  }

                  val resultCode = intent?.getIntExtra("RESULT_CODE", Activity.RESULT_CANCELED)
                      ?: Activity.RESULT_CANCELED
                  val data = intent?.let { IntentCompat.getParcelableExtra(it, "DATA", Intent::class.java) }

                  if (resultCode == Activity.RESULT_OK && data != null) {
                      initStockfish()
                      setupOverlay()
                      setupScreenCapture(resultCode, data)
                      updateStatus("Tahta aranıyor: oyunu başlangıç konumunda aç")
                  } else {
                      stopSelf()
                  }

                  return START_NOT_STICKY
              }

              /** Bildirimdeki durum yazısını günceller (tahta bulundu mu, motor hazır mı, öneri ne). */
              private fun updateStatus(text: String) {
                  if (text == lastStatus) return
                  lastStatus = text
                  val b = notificationBuilder ?: return
                  b.setContentText(text)
                  try {
                      getSystemService(NotificationManager::class.java)?.notify(1, b.build())
                  } catch (e: Exception) {
                      e.printStackTrace()
                  }
              }

              private fun setupOverlay() {
                  windowManager = getSystemService(WINDOW_SERVICE) as WindowManager

                  val box = GradientDrawable().apply {
                      setColor(0x6600FF55.toInt())       // yarı saydam parlak yeşil dolgu
                      setStroke(8, 0xFF00A524.toInt())   // koyu yeşil çerçeve
                  }
                  overlayView = FrameLayout(this).apply {
                      background = box
                      visibility = View.INVISIBLE
                  }

                  val layoutParams = WindowManager.LayoutParams(
                      1, 1,
                      if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O)
                          WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
                      else
                          @Suppress("DEPRECATION") WindowManager.LayoutParams.TYPE_PHONE,
                      WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                          WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE or
                          WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                          WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
                      PixelFormat.TRANSLUCENT
                  ).apply {
                      gravity = Gravity.TOP or Gravity.START
                      x = 0
                      y = 0
                  }

                  windowManager.addView(overlayView, layoutParams)
              }

              private fun setupScreenCapture(resultCode: Int, data: Intent) {
                  val metrics = DisplayMetrics()
                  @Suppress("DEPRECATION")
                  windowManager.defaultDisplay.getRealMetrics(metrics)
                  screenWidth = metrics.widthPixels
                  screenHeight = metrics.heightPixels

                  // Tahta ekran genişliğinde bir kare
                  boardSize = minOf(screenWidth, screenHeight)
                  boardTop = (screenHeight * BOARD_TOP_FRACTION).toInt()

                  val mpManager = getSystemService(Context.MEDIA_PROJECTION_SERVICE) as MediaProjectionManager
                  mediaProjection = mpManager.getMediaProjection(resultCode, data)

                  // Android 14+: createVirtualDisplay'den ÖNCE callback kaydedilmeli
                  mediaProjection?.registerCallback(object : MediaProjection.Callback() {
                      override fun onStop() { stopSelf() }
                  }, Handler(Looper.getMainLooper()))

                  captureThread = HandlerThread("capture").also { it.start() }
                  val handler = Handler(captureThread!!.looper)

                  imageReader = ImageReader.newInstance(screenWidth, screenHeight, PixelFormat.RGBA_8888, 2)
                  virtualDisplay = mediaProjection?.createVirtualDisplay(
                      "ChessCapture", screenWidth, screenHeight, metrics.densityDpi,
                      DisplayManager.VIRTUAL_DISPLAY_FLAG_AUTO_MIRROR, imageReader?.surface, null, null
                  )

                  imageReader?.setOnImageAvailableListener({ reader ->
                      val image = reader.acquireLatestImage() ?: return@setOnImageAvailableListener
                      try {
                          val now = SystemClock.elapsedRealtime()
                          if (now - lastProcessed < 1000 || analyzing.get()) {
                              return@setOnImageAvailableListener // finally image.close() yapar
                          }
                          lastProcessed = now

                          val plane = image.planes[0]
                          val pixelStride = plane.pixelStride
                          val rowPadding = plane.rowStride - pixelStride * screenWidth

                          val full = Bitmap.createBitmap(
                              screenWidth + rowPadding / pixelStride, screenHeight, Bitmap.Config.ARGB_8888
                          )
                          try {
                              full.copyPixelsFromBuffer(plane.buffer)
                              processFrame(full)
                          } finally {
                              full.recycle()
                          }
                      } catch (e: Exception) {
                          e.printStackTrace()
                      } finally {
                          image.close()
                      }
                  }, handler)
              }

              private fun processFrame(full: Bitmap) {
                  if (!recognizer.isCalibrated && !calibrateFromFrame(full)) {
                      updateStatus("Tahta aranıyor: oyunu başlangıç konumunda aç")
                      return
                  }
                  if (boardTop < 0 || boardTop + boardSize > full.height) return
                  val board = Bitmap.createBitmap(full, 0, boardTop, boardSize, boardSize)
                  try {
                      processBoard(board)
                  } finally {
                      board.recycle()
                  }
              }

              /** Önce varsayılan konumu dener; 5 denemede bir de çevresindeki konumları tarar. */
              private fun calibrateFromFrame(full: Bitmap): Boolean {
                  val base = (screenHeight * BOARD_TOP_FRACTION).toInt()
                  val step = maxOf(2, (screenHeight * 0.004f).toInt())
                  val offsets = ArrayList<Int>()
                  offsets.add(0)
                  if (calibTries % 5 == 0) {
                      for (i in 1..12) {
                          offsets.add(i * step)
                          offsets.add(-i * step)
                      }
                  }
                  calibTries++

                  for (off in offsets) {
                      val top = base + off
                      if (top < 0 || top + boardSize > full.height) continue
                      val b = Bitmap.createBitmap(full, 0, top, boardSize, boardSize)
                      val ok = recognizer.calibrate(b)
                      b.recycle()
                      if (ok) {
                          boardTop = top
                          boardFlipped = recognizer.flipped
                          candidate = null
                          stable = null
                          val side = if (boardFlipped) "siyah" else "beyaz"
                          Log.i("ChessHelper", "Kalibrasyon tamam. Tahta üst=$top, sen $side oynuyorsun")
                          updateStatus("Tahta bulundu ($side oynuyorsun). Hamle bekleniyor")
                          return true
                      }
                  }
                  return false
              }

              private fun processBoard(board: Bitmap) {
                  // Öneri kutusu ekranda görünüyorsa o kareyi okuma, sadece diğer karelerde değişiklik var mı bak
                  val of = overlayFile
                  val orank = overlayRank
                  if (of >= 0) {
                      val posUnderBox = recognizer.read(board, of, orank) ?: return
                      val st = stable
                      if (st != null && posUnderBox.sameExcept(st, of, orank)) return
                      hideOverlay() // gerçek bir değişiklik var: kutuyu kaldır, sonraki karelerden devam et
                      candidate = null
                      return
                  }

                  val pos = recognizer.read(board)
                  if (pos == null) {
                      unreadable++
                      if (unreadable >= 5) updateStatus("Tahta okunamıyor (taşlar tanınmadı)")
                      return
                  }
                  unreadable = 0

                  // İki ardışık karede aynı çıkmalı (animasyon filtresi)
                  if (pos.placement() != candidate?.placement()) {
                      candidate = pos
                      return
                  }

                  val prev = stable
                  if (prev != null && prev.placement() == pos.placement()) return // değişiklik yok
                  stable = pos

                  // Sıra kimde? Son hamleyi yapanın rengine bakılır
                  val userColor = if (recognizer.flipped) 'b' else 'w'
                  val turn: Char = if (prev == null) {
                      'w' // ilk okuma = başlangıç konumu, beyaz başlar
                  } else {
                      when (ChessPosition.moverColor(prev, pos)) {
                          'w' -> 'b'
                          'b' -> 'w'
                          else -> userColor
                      }
                  }

                  if (turn == userColor) {
                      updateStatus("Analiz ediliyor...")
                      getBestMoveFromStockfish(pos.fen(turn))
                  } else {
                      updateStatus("Rakibin hamlesi bekleniyor")
                  }
              }

              private fun initStockfish() {
                  engineExecutor.execute {
                      try {
                          val engine = File(applicationInfo.nativeLibraryDir, "libstockfish.so")
                          stockfishProcess = ProcessBuilder(engine.absolutePath)
                              .redirectErrorStream(true)
                              .start()
                          stockfishWriter = OutputStreamWriter(stockfishProcess?.outputStream)
                          stockfishReader = BufferedReader(InputStreamReader(stockfishProcess?.inputStream))

                          sendStockfishCommand("uci")
                          waitFor("uciok")
                          sendStockfishCommand("isready")
                          waitFor("readyok")
                      } catch (e: Exception) {
                          Log.e("ChessHelper", "Stockfish baslatilamadi", e)
                          updateStatus("Stockfish motoru başlatılamadı")
                      }
                  }
              }

              private fun waitFor(token: String) {
                  while (true) {
                      val line = stockfishReader?.readLine() ?: return
                      if (line.startsWith(token)) return
                  }
              }

              private fun sendStockfishCommand(command: String) {
                  try {
                      stockfishWriter?.write(command + "\n")
                      stockfishWriter?.flush()
                  } catch (e: Exception) {
                      e.printStackTrace()
                  }
              }

              private fun getBestMoveFromStockfish(fen: String) {
                  if (!analyzing.compareAndSet(false, true)) return

                  engineExecutor.execute {
                      try {
                          sendStockfishCommand("position fen $fen")
                          sendStockfishCommand("go depth 12")

                          while (true) {
                              val line = stockfishReader?.readLine() ?: break
                              if (line.startsWith("bestmove")) {
                                  val parts = line.split(" ")
                                  if (parts.size > 1) {
                                      val move = parts[1] // örn. "e2e4"
                                      updateStatus("Öneri: $move")
                                      Handler(Looper.getMainLooper()).post { updateOverlayPosition(move) }
                                  }
                                  break
                              }
                          }
                      } catch (e: Exception) {
                          e.printStackTrace()
                      } finally {
                          analyzing.set(false)
                      }
                  }
              }

              private fun hideOverlay() {
                  overlayFile = -1
                  overlayRank = -1
                  val view = overlayView ?: return
                  view.post { view.visibility = View.INVISIBLE }
              }

              // "e2e4" -> hedef kare (e4) üzerine yeşil kutuyu koyar
              private fun updateOverlayPosition(bestMove: String) {
                  if (bestMove.length < 4) return
                  val file = bestMove[2] - 'a'
                  val rank = bestMove[3] - '1'
                  if (file !in 0..7 || rank !in 0..7) return

                  val col = if (boardFlipped) 7 - file else file
                  val row = if (boardFlipped) rank else 7 - rank
                  val squareSize = boardSize / 8

                  val view = overlayView ?: return
                  try {
                      val params = view.layoutParams as WindowManager.LayoutParams
                      params.width = squareSize
                      params.height = squareSize
                      params.x = col * squareSize
                      params.y = boardTop + row * squareSize
                      view.visibility = View.VISIBLE
                      windowManager.updateViewLayout(view, params)
                      overlayFile = file
                      overlayRank = rank
                  } catch (e: Exception) {
                      e.printStackTrace()
                  }
              }

              private fun createNotificationChannel() {
                  if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                      val channel = NotificationChannel(
                          "CHESS_CHANNEL", "Satranç Servisi", NotificationManager.IMPORTANCE_LOW
                      )
                      getSystemService(NotificationManager::class.java)?.createNotificationChannel(channel)
                  }
              }

              override fun onDestroy() {
                  super.onDestroy()
                  overlayView?.let {
                      try { windowManager.removeView(it) } catch (e: Exception) { e.printStackTrace() }
                  }
                  imageReader?.close()
                  virtualDisplay?.release()
                  mediaProjection?.stop()
                  captureThread?.quitSafely()
                  sendStockfishCommand("quit")
                  engineExecutor.shutdownNow()
                  stockfishProcess?.destroy()
              }
          }
